tests/transactions/create-transfer.spec.ts
import { test, expect } from '../fixtures';

test.describe('Send money', () => {
  test.beforeEach(async ({ api, user }) => {
    // seed test data through the API, not the UI
    await api.createBeneficiary(user.id, { country: 'PK', method: 'BANK' });
  });

  test('customer can send a bank transfer @smoke @regression', async ({ sendMoneyPage, receiptPage }) => {
    await sendMoneyPage.open();
    await sendMoneyPage.enterAmount(250, 'GBP');
    await expect(sendMoneyPage.exchangeRate).toBeVisible();
    await sendMoneyPage.selectBeneficiary(0);
    await sendMoneyPage.confirmAndPay();

    await expect(receiptPage.status).toHaveText('Transaction created');
    await expect(receiptPage.reference).toHaveText(/^ACE\d{8}$/);
  });
});

queries/verify_transaction.sql
-- confirm the UI/API transaction landed correctly in the database
SELECT t.reference, t.status, t.send_amount, t.payout_amount,
       c.kyc_status, b.country
FROM   transactions t
JOIN   customers    c ON c.id = t.customer_id
JOIN   beneficiaries b ON b.id = t.beneficiary_id
WHERE  t.reference = @txnRef
  AND  c.kyc_status = 'VERIFIED';

-- catch duplicates created by double-click / retry
SELECT customer_id, send_amount, COUNT(*) AS dupes
FROM   transactions
WHERE  created_at > DATEADD(minute, -5, GETDATE())
GROUP BY customer_id, send_amount HAVING COUNT(*) > 1;

        

      

    

  


  
  

    

      
AI in QA

      
How I use AI every day

      
AI makes me faster. It doesn't make the decisions. Here's where it fits in my manual and automation work.

      

        

          
In manual testing

          
01
Requirement review
Ask AI to find gaps, contradictions and missing acceptance criteria in user stories before sprint planning.

          
02
Test case drafting
Turn a story into a first draft of positive, negative and boundary cases, then refine and add domain knowledge before it goes into TestRail.

          
03
Edge cases & test data
Generate realistic test data (names, IBANs, ID formats for different countries) and brainstorm edge cases for KYC and limits.

          
04
Bug reports
Structure clear Jira tickets: steps, expected vs actual, environment and impact, so developers can reproduce on the first try.

          
05
Testing AI itself
Tested the ACE AI chatbot for accuracy, hallucinations, fallback behaviour and conversation flow across account and corridor queries.

        

        

          
In automation

          
01
Planner agent
Playwright's Planner explores a live flow and drafts a structured test plan, which I review and approve.

          
02
Generator agent
Turns the approved plan into Playwright spec files that follow our Page Object Model and fixtures.

          
03
Healer agent
Repairs locators and steps broken by UI changes, so less time goes on maintenance and more on new coverage.

          
04
Code & query help
Draft Maestro flows, Postman test scripts and complex SQL joins, then debug failing tests from traces faster.

          
05
Failure triage
Summarise CI failures and spot patterns (flaky test vs real bug vs environment issue) before raising a defect.

        

      

      

        
        
Human in the loop, always. Every AI-generated plan, test, fix or bug report is reviewed by me before it's merged or filed. No real customer data ever goes into AI tools. In regulated fintech, that's not optional.
